import { useState } from "react";
import { Link, NavLink, Outlet, useNavigate, useLocation } from "react-router-dom";
import { LayoutDashboard, Smartphone, Package, ShoppingCart, Users, Settings, LogOut, Menu, X } from "lucide-react";

const navItems = [
  { to: "/admin", label: "Dashboard", icon: LayoutDashboard, end: true },
  { to: "/admin/products", label: "Products", icon: Smartphone },
  { to: "/admin/inventory", label: "Inventory", icon: Package },
  { to: "/admin/orders", label: "Orders", icon: ShoppingCart },
  { to: "/admin/customers", label: "Customers", icon: Users },
  { to: "/admin/settings", label: "Settings", icon: Settings },
];

export default function AdminLayout() {
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const navigate = useNavigate();
  const location = useLocation();

  const handleLogout = () => {
    localStorage.removeItem("techbox_admin");
    navigate("/admin/login");
  };

  return (
    <div className="tb-admin">
      {/* Mobile overlay */}
      {sidebarOpen && (
        <div className="tb-admin__overlay" onClick={() => setSidebarOpen(false)} />
      )}

      {/* Sidebar */}
      <aside className={`tb-admin__sidebar ${sidebarOpen ? "is-open" : ""}`}>
        <div className="tb-admin__sidebar-header">
          <Link to="/admin" className="tb-admin__brand">TECHBOX <span>Admin</span></Link>
          <button className="tb-admin__mobile-close" onClick={() => setSidebarOpen(false)}>
            <X size={20} />
          </button>
        </div>
        
        <nav className="tb-admin__nav">
          {navItems.map((item) => (
            <NavLink
              key={item.to}
              to={item.to}
              end={item.end}
              className="tb-admin__nav-link"
              onClick={() => setSidebarOpen(false)}
            >
              <item.icon size={18} />
              {item.label}
            </NavLink>
          ))}
        </nav>

        <div className="tb-admin__sidebar-footer">
          <button onClick={handleLogout} className="tb-admin__logout">
            <LogOut size={18} /> Logout
          </button>
        </div>
      </aside>

      {/* Main Content */}
      <div className="tb-admin__main">
        <header className="tb-admin__header">
          <div style={{ display: "flex", alignItems: "center", gap: "16px" }}>
            <button className="tb-admin__mobile-toggle" onClick={() => setSidebarOpen(true)}>
              <Menu size={24} />
            </button>
            <h1 className="tb-admin__page-title">
              {navItems.find((n) => n.to === location.pathname)?.label || "Admin"}
            </h1>
          </div>
          <div className="tb-admin__user">
            <div className="tb-admin__avatar">A</div>
            <span>Admin User</span>
          </div>
        </header>

        <main className="tb-admin__content">
          <Outlet />
        </main>
      </div>
    </div>
  );
}

import HomePage from "./pages/HomePage.jsx";
import ShopPage from "./pages/ShopPage.jsx";

import CartPage from "./pages/CartPage.jsx";
import CheckoutPage from "./pages/CheckoutPage.jsx";
import OrderConfirmationPage from "./pages/OrderConfirmationPage.jsx";

// Admin Imports
import AdminLayout from "./admin/layout/AdminLayout.jsx";
import AdminLoginPage from "./admin/pages/AdminLoginPage.jsx";
import DashboardPage from "./admin/pages/DashboardPage.jsx";
import AdminProductsPage from "./admin/pages/AdminProductsPage.jsx";

// Protected Route Component
function RequireAdminAuth({ children }) {
  const isAuthenticated = localStorage.getItem("techbox_admin") === "true";
  return isAuthenticated ? children : <Navigate to="/admin/login" replace />;
}

export default function App() {
  return (
    <CartProvider>
      <Routes>
        {/* Storefront Routes */}
        <Route path="/" element={<StorefrontWrapper />} />
        <Route path="/shop" element={<StorefrontWrapper />} />
        <Route path="/product/:slug" element={<StorefrontWrapper />} />
        <Route path="/cart" element={<StorefrontWrapper />} />
        <Route path="/checkout" element={<StorefrontWrapper />} />
        <Route path="/order-confirmation/:orderNumber" element={<StorefrontWrapper />} />
        <Route path="/contact" element={<StorefrontWrapper />} />
        <Route path="*" element={<StorefrontWrapper />} />

        {/* Admin Routes */}
        <Route path="/admin/login" element={<AdminLoginPage />} />
        <Route path="/admin" element={<RequireAdminAuth><AdminLayout /></RequireAdminAuth>}>
          <Route index element={<DashboardPage />} />
          <Route path="products" element={<AdminProductsPage />} />
          <Route path="products/new" element={<AdminProductFormPage />} />
          <Route path="inventory" element={<AdminInventoryPage />} />
          <Route path="orders" element={<AdminPlaceholderPage title="Orders Management" />} />
          <Route path="customers" element={<AdminPlaceholderPage title="Customer Database" />} />
          <Route path="settings" element={<AdminPlaceholderPage title="Store Settings" />} />
        </Route>
      </Routes>
    </CartProvider>
  );
}

function StorefrontWrapper() {
  const { toast } = useCart();
  return (
    <>
      <Navbar />
      <main>
        <Routes>
          <Route path="/" element={<HomePage />} />
          <Route path="/shop" element={<ShopPage />} />
          <Route path="/product/:slug" element={<ProductPage />} />
          <Route path="/cart" element={<CartPage />} />
          <Route path="/checkout" element={<CheckoutPage />} />
          <Route path="/order-confirmation/:orderNumber" element={<OrderConfirmationPage />} />
          <Route path="/contact" element={<ContactPage />} />
          <Route path="*" element={<NotFoundPage />} />
        </Routes>
      </main>
      <Footer />
      <CartToast toast={toast} onClose={() => {}} />
    </>
  );
}
