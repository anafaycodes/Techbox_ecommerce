// Reads techbox-bundle.txt, creates every === FILE: path === entry.
// Usage: node scaffold.js techbox-bundle.txt
const fs = require("fs");
const path = require("path");

const bundlePath = process.argv[2];
if (!bundlePath) {
  console.error("Usage: node scaffold.js <bundle-file>");
  process.exit(1);
}

const text = fs.readFileSync(bundlePath, "utf8");
const lines = text.split(/\r?\n/);
const fileMarker = /^=== FILE: (.+?) ===$/;

let current = null;
let buffer = [];
let count = 0;

function flush() {
  if (!current) return;
  const full = buffer.join("\n").replace(/^\n+|\n+$/g, "");
  const dir = path.dirname(current);
  if (dir && dir !== ".") fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(current, full + "\n", "utf8");
  console.log("wrote", current);
  count++;
  current = null;
  buffer = [];
}

for (const line of lines) {
  const m = line.match(fileMarker);
  if (m) {
    flush();
    current = m[1];
  } else if (current) {
    buffer.push(line);
  }
}
flush();
console.log(`\nDone. ${count} files written.`);

// Centralized API client. Real HTTP integration wires up in Phase 2.

const BASE_URL = import.meta.env.VITE_API_BASE_URL ?? "";
export const USE_MOCKS = import.meta.env.VITE_USE_MOCKS !== "false";

export async function apiGet(path) {
  if (USE_MOCKS) throw new Error(`Mock mode: ${path} not implemented yet.`);
  const res = await fetch(`${BASE_URL}${path}`, { credentials: "include" });
  if (!res.ok) throw new Error(`Request failed: ${res.status}`);
  return res.json();
}

export async function apiPost(path, body) {
  if (USE_MOCKS) throw new Error(`Mock mode: ${path} not implemented yet.`);
  const res = await fetch(`${BASE_URL}${path}`, {
    method: "POST",
    credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`Request failed: ${res.status}`);
  return res.json();
}

import { motion, useReducedMotion } from "framer-motion";

export default function ScrollReveal({ children, delay = 0, y = 20, as = "div" }) {
  const reduced = useReducedMotion();
  const MotionTag = motion[as] || motion.div;
  if (reduced) return <MotionTag>{children}</MotionTag>;
  return (
    <MotionTag
      initial={{ opacity: 0, y }}
      whileInView={{ opacity: 1, y: 0 }}
      viewport={{ once: true, margin: "-80px" }}
      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1], delay }}
    >
      {children}
    </MotionTag>
  );
}

          ))}

      )}
